# Forbidden Magic: Reforged — растения скверны (слайс 5) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Портировать листву, саженец с ростом скверного дерева, Чернильный цветок и Куст Теней; закрыть пропущенные loot table у блоков M4; добавить исследования `TAINTTREE`/`BLACKFLOWER` и их crucible-рецепты.

**Architecture:** Листва — подкласс `LeavesBlock` с `noAmbientSound()`, саженец — `VegetationBlock` + `BonemealableBlock` с код-генерацией дерева (`TaintTree.grow`), цветок — `VegetationBlock` с random-tick распространением, куст — `DoublePlantBlock`. Loot table — JSON в `data/forbiddenmagic/loot_table/blocks/`.

**Tech Stack:** NeoForge 26.3.0.57-beta, Java 25, ModDevGradle, Thaumcraft 4: Reforged 1.2, GameTest.

**Spec:** `docs/superpowers/specs/2026-10-20-forbidden-magic-reforged-design.md`.

## Граница слайса

Входит: `taint_leaves`, `taint_sapling`, `umbral_rose` (Чернильный цветок), `umbral_rose_bush` (Куст Теней); loot table для них и для пяти блоков M4; рост дерева; исследования и два crucible-рецепта.

Не входит: скверный плод (предмет не портирован), чернила из цветка (нужен `resource`), бонемил-распространение куста (записано отступлением), Арканный торт, Клетка Гнева.

## Global Constraints

- Хост не редактируется; патчи не меняются.
- Тексты и аспекты — из оригинала: `TAINTTREE` (TREE 4, TAINT 6, POISON 2, PLANT 3), `BLACKFLOWER` (PLANT 3, SENSES 2, DARKNESS 4).

- [ ] **Step 3: `TaintTree`**

```java
package dev.visfox.forbiddenreforged;

import net.minecraft.core.BlockPos;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.util.RandomSource;
import net.minecraft.world.level.block.state.BlockState;

public final class TaintTree {
    private TaintTree() {
    }

    public static boolean grow(ServerLevel level, BlockPos pos, RandomSource random) {
        int height = 4 + random.nextInt(3);
        if (pos.getY() + height + 2 >= level.getMaxY()) {
            return false;
        }
        for (int y = 0; y <= height; y++) {
            if (!level.getBlockState(pos.above(y)).canBeReplaced()) {
                return false;
            }
        }
        BlockState log = FmBlocks.TAINT_LOG.get().defaultBlockState();
        for (int y = 0; y < height; y++) {
            level.setBlockAndUpdate(pos.above(y), log);
        }
        BlockState leaves = FmBlocks.TAINT_LEAVES.get().defaultBlockState();
        for (int layer = 0; layer < 3; layer++) {
            int radius = layer == 0 ? 2 : 1;
            int y = height - 1 + layer;
            for (int dx = -radius; dx <= radius; dx++) {
                for (int dz = -radius; dz <= radius; dz++) {
                    if (Math.abs(dx) == radius && Math.abs(dz) == radius && layer == 2) {
                        continue;
                    }
                    BlockPos target = pos.above(y).offset(dx, 0, dz);
                    if (level.getBlockState(target).canBeReplaced()) {
                        level.setBlockAndUpdate(target, leaves);
                    }
                }
            }
        }
        level.setBlockAndUpdate(pos.above(height), leaves);
        return true;
    }
}
```

```java
package dev.visfox.forbiddenreforged;

import net.minecraft.core.BlockPos;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.util.RandomSource;
import net.minecraft.world.level.Level;
import net.minecraft.world.level.LevelReader;
import net.minecraft.world.level.block.BonemealableBlock;
import net.minecraft.world.level.block.VegetationBlock;
import net.minecraft.world.level.block.state.BlockState;

public class TaintSaplingBlock extends VegetationBlock implements BonemealableBlock {
    public TaintSaplingBlock(Properties properties) {
        super(properties);
    }

    @Override
    protected void randomTick(BlockState state, ServerLevel level, BlockPos pos, RandomSource random) {
        if (random.nextInt(10) == 0 && level.getMaxLocalRawBrightness(pos.above()) >= 9) {
            grow(level, pos, random);
        }
    }

    private void grow(ServerLevel level, BlockPos pos, RandomSource random) {
        if (!TaintTree.grow(level, pos, random)) {
            level.setBlockAndUpdate(pos, defaultBlockState());
        }
    }

    @Override
    public boolean isValidBonemealTarget(LevelReader level, BlockPos pos, BlockState state) {
        return true;
    }

    @Override
    public boolean isBonemealSuccess(Level level, RandomSource random, BlockPos pos, BlockState state) {
        return true;
    }

    @Override
    public void performBonemeal(ServerLevel level, RandomSource random, BlockPos pos, BlockState state) {
        grow(level, pos, random);
    }
}
```

```java
    @Override
    protected void randomTick(BlockState state, ServerLevel level, BlockPos pos, RandomSource random) {
        if (random.nextInt(10) <= 2) {
            spread(level, pos, random);
        }
    }

    private static void spread(ServerLevel level, BlockPos pos, RandomSource random) {
        int nearby = 0;
        for (BlockPos cursor : BlockPos.betweenClosed(pos.offset(-2, -1, -2), pos.offset(2, 1, 2))) {
            if (level.getBlockState(cursor).is(FmBlocks.UMBRAL_ROSE.get()) && ++nearby >= 10) {
                return;
            }
        }
        BlockPos target = pos.offset(random.nextInt(3) - 1, random.nextInt(2) - random.nextInt(2), random.nextInt(3) - 1);
        if (level.getBlockState(target).canBeReplaced() && level.getBlockState(target.below()).is(BlockTags.DIRT)) {
            level.setBlockAndUpdate(target, FmBlocks.UMBRAL_ROSE.get().defaultBlockState());
        }
    }
```
